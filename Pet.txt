CLASS Pet
    ATTR name, age, breed
    CONSTRUCTOR(name, age, breed)

CLASS Dog EXTENDS Pet
    ATTR isTrained
    CONSTRUCTOR(name, age, breed, isTrained)

CLASS Cat EXTENDS Pet
    ATTR color
    CONSTRUCTOR(name, age, breed, color)

CLASS Bird EXTENDS Pet
    ATTR canTalk
    CONSTRUCTOR(name, age, breed, canTalk)

bird = NEW Bird("Mittu", 1, "Parrot", true)
