interface LightBehavior { void activate(); }

LightBehavior motionTrigger = () -> System.out.println("Light ON with motion sensor pattern");
LightBehavior timeTrigger   = () -> System.out.println("Light ON with evening dim mode");
LightBehavior voiceTrigger  = () -> System.out.println("Light ON with voice-activated brightness");
motionTrigger.activate(); timeTrigger.activate(); voiceTrigger.activate();

class Product { int price; int rating; int discount; }
List<Product> products = getProductList();
products.sort((p1, p2) -> p1.price - p2.price);
products.sort((p1, p2) -> p2.rating - p1.rating);
products.sort((p1, p2) -> p2.discount - p1.discount);

class Alert { String type; }
List<Alert> alerts = getAlerts();
