INTERFACE Searchable
    METHOD search(keyword)

INTERFACE Printable
    METHOD printReport()

CLASS PatientReport IMPLEMENTS Searchable, Printable
    ATTR patientName, diagnosis
    CONSTRUCTOR(patientName, diagnosis)

    METHOD search(keyword)
        IF keyword IN diagnosis OR keyword IN patientName
            PRINT "Found in Patient Report"

    METHOD printReport()
        PRINT "Patient Report: " + patientName + " - " + diagnosis

CLASS DoctorReport IMPLEMENTS Searchable, Printable
    ATTR doctorName, specialization
    CONSTRUCTOR(doctorName, specialization)

    METHOD search(keyword)
        IF keyword IN specialization OR keyword IN doctorName
            PRINT "Found in Doctor Report"

    METHOD printReport()
        PRINT "Doctor Report: " + doctorName + " - " + specialization

// Polymorphism Example
CREATE report1 = NEW PatientReport("Shiva", "Fever")
CREATE report2 = NEW DoctorReport("Dr. Mehta", "Cardiology")

CREATE reports = [report1, report2]

FOR EACH r IN reports
    r.search("Fever")
    r.printReport()
